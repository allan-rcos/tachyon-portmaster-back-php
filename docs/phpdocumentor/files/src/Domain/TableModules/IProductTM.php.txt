namespace Domain\TableModules;

use Domain\Enums\RiskClass;
use Domain\Models\IProduct;
use Shared\Exceptions\Result;

/**
 * The rules a product must satisfy to exist.
 *
 * Nothing else validates a product: not the controller, not the repository. A
 * use case asks for one and either gets it or gets the reason it cannot have
 * one.
 *
 * @see IProduct What gets built.
 * @see \Domain\TableModules\Interno\ProductTM The implementation.
 *
 * @license {@link https://opensource.org/licenses/MIT MIT}
 * @copyright 2026 Tachyon
 * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
 *
 * @since 0.0.1
 *
 * @version 0.0.1
 */
interface IProductTM
{
    /**
     * Builds a new product, assigning it an id.
     *
     * @param  string  $name  Commercial name; required, at most 255 characters.
     * @param  float  $density  Kilograms per litre; must be greater than zero.
     * @param  RiskClass  $riskClass  UN classification, or `None`.
     * @return Result<IProduct> A 422 failure listing every field that broke a
     *                          rule, not just the first.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function create(string $name, float $density, RiskClass $riskClass): Result;

    /**
     * Builds the updated state of an existing product, re-running every rule.
     *
     * The id is supplied rather than generated — an update keeps its identity.
     *
     * @param  string  $id  Id of the product being updated.
     * @param  string  $name  Commercial name; required, at most 255 characters.
     * @param  float  $density  Kilograms per litre; must be greater than zero.
     * @param  RiskClass  $riskClass  UN classification, or `None`.
     * @return Result<IProduct> A 422 failure listing every field that broke a
     *                          rule, not just the first.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function update(string $id, string $name, float $density, RiskClass $riskClass): Result;
}
