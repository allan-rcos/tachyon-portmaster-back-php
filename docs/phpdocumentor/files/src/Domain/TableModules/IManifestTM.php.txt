namespace Domain\TableModules;

use Domain\Models\IManifestCargo;
use Domain\Models\IManifestChange;
use Domain\Models\IContainer;
use Domain\Models\IProduct;
use Shared\Exceptions\Result;

/**
 * Moving cargo in and out of a container.
 *
 * Loading and unloading live here rather than in {@see IContainerTM} because
 * each changes the container *and* its cargo together, and the two must agree:
 * a container's weight is the sum of its cargo lines, so nothing may move one
 * without the other. Both methods answer with a single
 * {@see IManifestChange} carrying every consequence, and the use case persists
 * the parts without deciding anything.
 *
 * The caller supplies the product's existing cargo line rather than this module
 * fetching it — the domain reads no storage.
 *
 * @see IManifestChange What comes back.
 * @see \Domain\TableModules\Interno\ManifestTM The implementation.
 *
 * @license {@link https://opensource.org/licenses/MIT MIT}
 * @copyright 2026 Tachyon
 * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
 *
 * @since 0.0.1
 *
 * @version 0.0.1
 */
interface IManifestTM
{
    /**
     * Loads a quantity of a product into the container.
     *
     * Computes the added weight from the product's density, checks the status
     * and the remaining capacity, transitions `Empty` to `Loading`, and names
     * the telemetry event to record.
     *
     * @param  IContainer  $container  Current container state.
     * @param  IProduct  $product  What is being loaded; its density converts
     *                             quantity to weight.
     * @param  float  $quantity  Units to add; must be greater than zero.
     * @param  ?IManifestCargo  $current  The product's existing cargo line, or
     *                                    null if it is not yet aboard.
     * @return Result<IManifestChange> A 422 failure on a non-positive quantity;
     *                                 a 409 when the container is not accepting
     *                                 cargo or the load exceeds its capacity.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function load(IContainer $container, IProduct $product, float $quantity, ?IManifestCargo $current): Result;

    /**
     * Unloads a quantity of a product from the container.
     *
     * Requires status `Loading`, and refuses to take out more than went in.
     * When the container empties, the status returns to `Empty` and the whole
     * manifest is cleared rather than one line removed.
     *
     * @param  IContainer  $container  Current container state.
     * @param  IProduct  $product  What is being unloaded.
     * @param  float  $quantity  Units to remove; must be greater than zero.
     * @param  ?IManifestCargo  $current  The product's existing cargo line, or
     *                                    null if it is not aboard.
     * @return Result<IManifestChange> A 422 failure on a non-positive quantity;
     *                                 a 409 when the container is not `Loading`
     *                                 or holds less than was asked for.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function unload(IContainer $container, IProduct $product, float $quantity, ?IManifestCargo $current): Result;
}
