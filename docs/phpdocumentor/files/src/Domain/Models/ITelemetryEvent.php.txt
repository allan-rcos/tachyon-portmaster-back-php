declare(strict_types=1);

namespace Domain\Models;

/**
 * A container telemetry event type — **system metadata**, same family as
 * {@see IPermission} and {@see IMarkerGroup}.
 *
 * Telemetry is the audit trail, and an audit trail grows with the system: every
 * new operation worth recording adds an event type. It is queryable and fully
 * rebuildable from code on restart, so it satisfies the metadata test and stops
 * being an enum.
 *
 * Contrast with {@see \Domain\Enums\ContainerStatus} and
 * {@see \Domain\Enums\RiskClass}: those are closed sets fixed by the business,
 * so they remain enums.
 *
 * @see IPermission Why this family carries no label or description.
 * @see \Infra\Repository\ITelemetryEventRepository Where these are stored.
 *
 * @license {@link https://opensource.org/licenses/MIT MIT}
 * @copyright 2026 Tachyon
 * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
 *
 * @since 0.0.1
 *
 * @version 0.0.1
 */
interface ITelemetryEvent
{
    /**
     * @var string The stable identifier persisted in the telemetry log
     *             (e.g. `load`).
     *
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     */
    public string $slug {
        get;
    }

    /**
     * Registry index, assigned on registration. A lookup handle only, never
     * persisted — and, like {@see IPermission::$id}, only stable while the
     * `MEMORY` table holding the registry lives.
     *
     * @var int Registry index; zero means built but not yet registered.
     *
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     */
    public int $id {
        get;
    }

    /**
     * The same event carrying the registry index.
     *
     * Used by the repository at registration time, since the id is not knowable
     * when the domain builds the event.
     *
     * @param  int  $id  The index the registry assigned.
     * @return self A copy carrying it; the receiver is unchanged.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function withId(int $id): self;
}
