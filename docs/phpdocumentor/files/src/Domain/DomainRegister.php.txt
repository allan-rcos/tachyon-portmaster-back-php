declare(strict_types=1);

namespace Domain;

use Domain\Config\DomainConfig;
use Domain\Interno\DomainProvider;

/**
 * Composition entry point for the domain layer — the innermost register.
 *
 * The end of the chain: `ApiRegister` → `AppRegister` → `InfraRegister` →
 * here. Nothing is injected, because the domain depends on nothing. Even the
 * password hasher its table modules need is a domain implementation, built
 * privately by the provider.
 *
 * @see IDomainProvider What comes back.
 * @see docs/adr/0006-layered-providers-per-feature.md Why the wiring is hand-written.
 *
 * @license {@link https://opensource.org/licenses/MIT MIT}
 * @copyright 2026 Tachyon
 * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
 *
 * @since 0.0.1
 *
 * @version 0.0.1
 */
final class DomainRegister
{
    /**
     * Builds the domain provider for one worker.
     *
     * @param  DomainConfig  $config  Snowflake epoch and cluster id.
     * @param  int  $serverId  The OpenSwoole worker id, which becomes the
     *                         Snowflake machine id — what makes concurrent
     *                         workers generate non-colliding ids without
     *                         coordinating.
     * @return IDomainProvider The layer's factory surface.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public static function execute(DomainConfig $config, int $serverId): IDomainProvider
    {
        return new DomainProvider($config, $serverId);
    }
}
