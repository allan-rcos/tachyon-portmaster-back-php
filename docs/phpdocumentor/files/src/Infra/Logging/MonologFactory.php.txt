declare(strict_types=1);

namespace Infra\Logging;

use Infra\Config\ServerConfigLogLevel;
use Monolog\Formatter\JsonFormatter;
use Monolog\Handler\StreamHandler;
use Monolog\Logger;

/**
 * Builds the application's logger.
 *
 * The single place Monolog is wired: JSON to stdout, which is what a container
 * runtime expects to collect. Nothing else configures a handler, so changing
 * where logs go is a change here alone.
 *
 * @see ILogger What it produces, behind the port.
 * @see MonologAdapter The concrete implementation it wraps the logger in.
 * @see \Infra\Config\LogConfig Where the level comes from.
 *
 * @license {@link https://opensource.org/licenses/MIT MIT}
 * @copyright 2026 Tachyon
 * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
 *
 * @since 0.0.1
 *
 * @version 0.0.1
 */
class MonologFactory
{
    /**
     * Assembles a stdout JSON logger at the given level.
     *
     * One handler and no fallback: if stdout is unavailable there is nowhere
     * else this would sensibly write.
     *
     * @param  string  $name  The root channel; services rebrand it with
     *                        {@see ILogger::withChannel()}.
     * @param  ServerConfigLogLevel  $level  The floor below which lines are
     *                                       dropped.
     * @return ILogger The application logger, behind the port.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public static function create(
        string $name = 'swoole-api',
        ServerConfigLogLevel $level = ServerConfigLogLevel::INFO
    ): ILogger {
        $logger = new Logger($name);

        $handler = new StreamHandler('php://stdout', $level->value);

        $handler->setFormatter(new JsonFormatter());

        $logger->pushHandler($handler);

        return new MonologAdapter($logger);
    }
}
