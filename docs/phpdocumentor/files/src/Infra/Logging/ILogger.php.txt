declare(strict_types=1);

namespace Infra\Logging;

/**
 * The application's logging port.
 *
 * Deliberately narrower than PSR-3: four levels, a channel and a context bag,
 * which is everything this codebase actually uses. Nothing outside
 * {@see MonologAdapter} knows a logging library is involved.
 *
 * **Nothing here returns a {@see \Shared\Exceptions\Result}.** Logging is not
 * allowed to fail a request, so a broken handler is swallowed rather than
 * reported — a caller has nothing useful to do with the news that a log line did
 * not get written.
 *
 * @see MonologAdapter The implementation.
 * @see MonologFactory What builds one.
 *
 * @license {@link https://opensource.org/licenses/MIT MIT}
 * @copyright 2026 Tachyon
 * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
 *
 * @since 0.0.1
 *
 * @version 0.0.1
 */
interface ILogger
{
    /**
     * Records detail useful while diagnosing, not in ordinary operation.
     *
     * @param  string  $message  What happened.
     * @param  array<string, mixed>  $context  Structured detail to go alongside.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function debug(string $message, array $context = []): void;

    /**
     * Records something worth knowing that is working as intended.
     *
     * Also where a client's own mistake goes — a 404 is not the server's
     * problem.
     *
     * @param  string  $message  What happened.
     * @param  array<string, mixed>  $context  Structured detail to go alongside.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function info(string $message, array $context = []): void;

    /**
     * Records something unexpected that was nonetheless handled.
     *
     * @param  string  $message  What happened.
     * @param  array<string, mixed>  $context  Structured detail to go alongside.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function warn(string $message, array $context = []): void;

    /**
     * Records a failure — the level every 500 in this codebase is logged at.
     *
     * @param  string  $message  What happened.
     * @param  array<string, mixed>  $context  Structured detail to go alongside,
     *                                         conventionally including the
     *                                         driver's or exception's own
     *                                         message.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function error(string $message, array $context = []): void;

    /**
     * A logger writing under a different channel name.
     *
     * A new instance, not a mutation, which is what lets one injected logger be
     * rebranded by every service that takes it without any of them affecting the
     * others. Services do this in their constructor, so a line can be traced to
     * the component that wrote it.
     *
     * @param  string  $name  The channel, conventionally the service's own name
     *                        in kebab-case.
     * @return ILogger A logger on that channel.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function withChannel(string $name): ILogger;

    /**
     * Attaches a value to every line logged for the rest of the current
     * request.
     *
     * Not a property of the logger: the value is filed against the running
     * coroutine, so *every* logger in that request picks it up, whatever channel
     * it is on and whoever holds it. That is what lets a request id be set once
     * at the edge and appear on every line the request goes on to produce.
     *
     * Outside a coroutine it has nothing to write to, and warns rather than
     * failing.
     *
     * @param  string  $key  What to file it under.
     * @param  string  $value  The value.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function setContext(string $key, string $value): void;
}
