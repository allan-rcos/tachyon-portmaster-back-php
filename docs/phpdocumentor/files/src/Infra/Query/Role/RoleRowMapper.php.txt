declare(strict_types=1);

namespace Infra\Query\Role;

use Domain\ID\Base62;

/**
 * Maps a `roles` row (with the DQL's `user_count` aggregate and JSON
 * `permissions` column) into a {@see RoleViewItem}. Shared by the role DQLs.
 *
 * Every field is coerced rather than trusted, so a column that is missing, null
 * or of an unexpected type yields a usable default instead of a type error
 * mid-hydration. A row without the `user_count` aggregate therefore maps
 * cleanly, reporting zero.
 *
 * @see RoleViewItem What it produces.
 * @see \Infra\Query\Interno\ListRolesDQL One of its callers.
 *
 * @license {@link https://opensource.org/licenses/MIT MIT}
 * @copyright 2026 Tachyon
 * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
 *
 * @since 0.0.1
 *
 * @version 0.0.1
 */
final class RoleRowMapper
{
    /**
     * Builds one view item from a row, encoding the stored id back to Base62.
     *
     * @param  array<string, mixed>  $row  As the driver returned it, including
     *                                     the query's `user_count` aggregate.
     * @return RoleViewItem The mapped item; never null, even for a row that
     *                      carried nothing recognisable.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public static function item(array $row): RoleViewItem
    {
        $id = is_numeric($row['id'] ?? null) ? (int) $row['id'] : 0;
        $name = is_scalar($row['name'] ?? null) ? (string) $row['name'] : '';
        $userCount = is_numeric($row['user_count'] ?? null) ? (int) $row['user_count'] : 0;

        return new RoleViewItem(
            id: Base62::encode($id),
            name: $name,
            userCount: $userCount,
            permissions: self::permissions($row['permissions'] ?? null),
        );
    }

    /**
     * The stored permission slugs. Unrecognised ones are kept rather than
     * dropped — the registry is a runtime catalogue, not a validator of history.
     *
     * Non-string entries *are* dropped, which is a different thing: a slug this
     * build does not know is still a slug, whereas a number in that column is
     * not.
     *
     * @param  mixed  $raw  The `permissions` column, expected to be a JSON
     *                      array; anything else reads as no permissions at all.
     * @return list<string> The slugs, re-indexed.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     */
    private static function permissions(mixed $raw): array
    {
        $decoded = is_string($raw) ? json_decode($raw, true) : [];
        if (!is_array($decoded)) {
            return [];
        }

        return array_values(array_filter($decoded, 'is_string'));
    }
}
