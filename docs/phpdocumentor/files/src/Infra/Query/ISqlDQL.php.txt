declare(strict_types=1);

namespace Infra\Query;

/**
 * The SQL-backend facet of a query: how the DQL compiles to a {@see SqlQuery}.
 *
 * A different backend adds its own facet (e.g. `IMongoDQL` with `toMongo()`) and
 * {@see IDQL} extends it too, forcing every query to implement the new method —
 * even if only to return a failure. The matching runner (SqlQueryRepository,
 * MongoQueryRepository) consumes the facet it understands.
 *
 * @see IDQL The full query contract, which extends this one.
 * @see SqlQuery What compilation produces.
 * @see \Infra\Query\Interno\SqlQueryRepository The runner that consumes this facet.
 *
 * @license {@link https://opensource.org/licenses/MIT MIT}
 * @copyright 2026 Tachyon
 * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
 *
 * @since 0.0.1
 *
 * @version 0.0.1
 */
interface ISqlDQL
{
    /**
     * Compiles the query into the statement and bindings to execute.
     *
     * Called once per run, so a DQL is free to build its SQL from whatever
     * filters it was constructed with rather than keeping a prepared string.
     *
     * @return SqlQuery The statement and its bound values, ready for the runner.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function toSql(): SqlQuery;
}
