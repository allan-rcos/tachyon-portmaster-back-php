declare(strict_types=1);

namespace Infra\Query\Container;

/**
 * One container together with what it carries and what has happened to it.
 *
 * Composes rather than flattens: the container keeps its own
 * {@see ContainerViewItem}, so the same shape a plain listing returns is
 * embedded here unchanged.
 *
 * @see ContainerSummaryListView The paged result these make up.
 * @see \Infra\Query\Interno\ListContainerSummariesDQL What builds one.
 *
 * @license {@link https://opensource.org/licenses/MIT MIT}
 * @copyright 2026 Tachyon
 * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
 *
 * @since 0.0.1
 *
 * @version 0.0.1
 */
final readonly class ContainerSummaryViewItem
{
    /**
     * @param  ContainerViewItem  $container  The container itself.
     * @param  list<CargoItemView>  $manifest  Its cargo lines; empty when it
     *                                         carries nothing.
     * @param  list<TelemetryLogView>  $recentLogs  The newest entries of its
     *                                              history, capped by the query
     *                                              rather than the whole log.
     *                                              The order within the list is
     *                                              the aggregation's and is not
     *                                              guaranteed.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     */
    public function __construct(
        public ContainerViewItem $container,
        public array $manifest,
        public array $recentLogs,
    ) {
    }
}
