declare(strict_types=1);

namespace Infra\Query\Container;

use Domain\Enums\ContainerStatus;

/**
 * A single container read record — the container itself, without its cargo.
 *
 * Read-side DTO, built only by the container DQLs through
 * {@see ContainerRowMapper}. What it carries is loaded separately, into
 * {@see ContainerSummaryViewItem}.
 *
 * @see ContainerListView The paged result these make up.
 * @see ContainerRowMapper What builds one from a row.
 *
 * @license {@link https://opensource.org/licenses/MIT MIT}
 * @copyright 2026 Tachyon
 * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
 *
 * @since 0.0.1
 *
 * @version 0.0.1
 */
final readonly class ContainerViewItem
{
    /**
     * @param  string  $id  Base62, already encoded for the caller.
     * @param  string  $code  The container's human-facing identifier.
     * @param  float  $currentWeight  What it holds now, as stored on the row
     *                                rather than resummed from the manifest.
     * @param  float  $maxCapacity  What it may hold.
     * @param  ContainerStatus  $status  Resolved into the enum; an unrecognised
     *                                   stored value degrades to
     *                                   {@see ContainerStatus::Empty}.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     */
    public function __construct(
        public string $id,
        public string $code,
        public float $currentWeight,
        public float $maxCapacity,
        public ContainerStatus $status,
    ) {
    }
}
