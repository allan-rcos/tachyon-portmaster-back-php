namespace Infra\Database\Pool;

use PDO;
use Shared\Exceptions\Result;

/**
 * Lends out database connections and takes them back.
 *
 * The API runs on coroutines, so a connection cannot simply be a per-process
 * singleton: several requests are in flight at once and each needs its own while
 * it holds a transaction open. Borrowing bounds how many exist at all, and
 * returning is what lets the next coroutine proceed rather than block.
 *
 * **Every {@see get()} owes a {@see put()}**, on the failure paths too — a lease
 * that is never returned is a connection permanently removed from circulation,
 * and enough of those deadlock the worker. Callers pair them in a `finally`.
 *
 * @see Interno\OpenSwoolePDOClientPool The implementation.
 * @see \Infra\Query\Interno\SqlQueryRepository A caller: leases per query.
 *
 * @license {@link https://opensource.org/licenses/MIT MIT}
 * @copyright 2026 Tachyon
 * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
 *
 * @since 0.0.1
 *
 * @version 0.0.1
 */
interface IPDOPool
{
    /**
     * Borrows a connection, waiting for one when they are all out.
     *
     * @return Result<PDO> The connection, which the caller now owes back; a 500
     *                     failure when none could be obtained.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function get(): Result;

    /**
     * Returns a borrowed connection to circulation.
     *
     * @param  PDO  $pdo  The connection a previous {@see get()} handed out.
     * @return Result<null> Void on success; a 500 failure when it could not be
     *                      returned.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function put(PDO $pdo): Result;
}
