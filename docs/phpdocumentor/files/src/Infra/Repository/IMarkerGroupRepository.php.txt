declare(strict_types=1);

namespace Infra\Repository;

use Domain\Models\IMarkerGroup;
use Ds\Seq;
use Shared\Exceptions\Result;

/**
 * The marker-group registry: the runtime catalogue of every marker group a
 * feature declared.
 *
 * Same metadata contract as {@see IPermissionRepository}, over the
 * `marker_groups` table — see there for why a catalogue derivable from the code
 * is a registry rather than a persistence repository, and for why the reads
 * return bare values instead of {@see Result}.
 *
 * What is particular here is who reads it: every marker operation does, to turn
 * the group slug a caller names into the index actually stored on the marker
 * row. A group that no worker registered therefore makes
 * {@see IMarkerRepository::set()} fail with a 404 rather than writing a marker
 * nothing could later find.
 *
 * @see IMarkerGroup What is registered.
 * @see IMarkerRepository The markers filed under these groups.
 * @see \Infra\Repository\Interno\MarkerGroupRegistry The implementation.
 * @see docs/adr/0002-metadata-registries-in-the-database.md Why this lives in the database.
 *
 * @license {@link https://opensource.org/licenses/MIT MIT}
 * @copyright 2026 Tachyon
 * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
 *
 * @since 0.0.1
 *
 * @version 0.0.1
 */
interface IMarkerGroupRepository
{
    /**
     * Registers a group and returns it carrying its assigned registry index.
     *
     * Idempotent by slug, which is what lets every worker declare the same
     * groups at boot without duplicating them.
     *
     * @param  IMarkerGroup  $group  Built by the table module, with `id = 0`.
     * @return Result<IMarkerGroup> The registered group, carrying its index; a
     *                              500 failure when the write fails.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function add(IMarkerGroup $group): Result;

    /**
     * The registered group for a slug.
     *
     * @param  string  $slug  Name the feature declared the group under.
     * @return ?IMarkerGroup Null when the slug was never registered.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function getBySlug(string $slug): ?IMarkerGroup;

    /**
     * The registered group for a registry index.
     *
     * @param  int  $id  The index assigned at registration — what the marker row
     *                   stores.
     * @return ?IMarkerGroup Null when the index is unknown.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function getById(int $id): ?IMarkerGroup;

    /**
     * Every registered group, in registration order.
     *
     * @return Seq<IMarkerGroup> The whole catalogue; empty before any worker has
     *                           registered anything.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function all(): Seq;

    /**
     * Whether a slug has been registered.
     *
     * @param  string  $slug  Name the feature declared the group under.
     * @return bool True when it is in the catalogue.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function has(string $slug): bool;
}
