namespace Infra\Repository;

use Domain\Models\IRole;
use Shared\Exceptions\Result;

/**
 * Persistence for roles — the write side.
 *
 * Stores what the table module already validated; it does not validate. Every
 * method enlists in the caller's open transaction, so a use case that rolls back
 * undoes these too.
 *
 * A role carries the permission slugs it grants, so writing a role is what
 * changes an authorization decision. Which *users* hold a role is the other
 * side of the `user_roles` pivot and is written through
 * {@see IUserRepository::syncRoles()}; only reading it lives here.
 *
 * @see IRole What is stored.
 * @see \Domain\TableModules\IRoleTM Validates before anything reaches here.
 * @see \Infra\Repository\Interno\SqlRoleRepository The implementation.
 *
 * @license {@link https://opensource.org/licenses/MIT MIT}
 * @copyright 2026 Tachyon
 * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
 *
 * @since 0.0.1
 *
 * @version 0.0.1
 */
interface IRoleRepository
{
    /**
     * Loads a role by id.
     *
     * @param  string  $id  Base62 id as it travels the application.
     * @return Result<IRole> A 404 failure when no row matched; a 500 when the
     *                       read itself failed, which includes an id too
     *                       malformed to decode.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function findById(string $id): Result;

    /**
     * Loads every role assigned to a user, joined through the `user_roles`
     * pivot.
     *
     * This is the read the authorization middleware makes on each request, which
     * is why it is one join rather than a role lookup per assignment.
     *
     * @param  string  $userId  Base62 id of the user.
     * @return Result<IRole[]> The user's roles, empty when they hold none — a
     *                         user without roles is an ordinary answer, not a
     *                         404. A 500 failure when the read broke.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function findByUserId(string $userId): Result;

    /**
     * Writes a new role.
     *
     * @param  IRole  $role  Already validated by the table module.
     * @return Result<null> Void on success; a 500 failure on any write error,
     *                      including a constraint the database rejected.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function insert(IRole $role): Result;

    /**
     * Overwrites an existing role, matched on its id.
     *
     * The permission list is replaced wholesale, so every user holding the role
     * is affected on their next request.
     *
     * @param  IRole  $role  The new state, already validated.
     * @return Result<null> Void on success; a 500 failure on a write error.
     *                      Matching no row is *not* a failure — callers that
     *                      care load the role first.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function update(IRole $role): Result;

    /**
     * Removes a role.
     *
     * Its assignments go with it — `user_roles` cascades on delete — so users
     * holding it simply stop holding it.
     *
     * @param  string  $id  Base62 id.
     * @return Result<null> Void on success; a 500 failure on a write error.
     *                      Matching no row is *not* a failure — the use case
     *                      loads the role first, and that is where the 404 comes
     *                      from.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function delete(string $id): Result;
}
