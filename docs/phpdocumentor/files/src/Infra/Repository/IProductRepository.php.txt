namespace Infra\Repository;

use Domain\Models\IProduct;
use Shared\Exceptions\Result;

/**
 * Persistence for products — the write side.
 *
 * Stores what the table module already validated; it does not validate. Every
 * method enlists in the caller's open transaction, so a use case that rolls
 * back undoes these too.
 *
 * Listing and searching are not here: those go through
 * {@see \Infra\Query\IQueryRepository}, which returns views rather than domain
 * models.
 *
 * @see IProduct What is stored.
 * @see \Domain\TableModules\IProductTM Validates before anything reaches here.
 * @see \Infra\Repository\Interno\SqlProductRepository The implementation.
 *
 * @license {@link https://opensource.org/licenses/MIT MIT}
 * @copyright 2026 Tachyon
 * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
 *
 * @since 0.0.1
 *
 * @version 0.0.1
 */
interface IProductRepository
{
    /**
     * Loads a product by id.
     *
     * @param  string  $id  Base62 id as it travels the application.
     * @return Result<IProduct> A 404 failure when no row matched; a 500 when the
     *                          read itself failed, which includes an id too
     *                          malformed to decode.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function findById(string $id): Result;

    /**
     * Writes a new product.
     *
     * @param  IProduct  $product  Already validated by the table module.
     * @return Result<null> Void on success; a 500 failure on any write error,
     *                      including a constraint the database rejected.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function insert(IProduct $product): Result;

    /**
     * Overwrites an existing product, matched on its id.
     *
     * @param  IProduct  $product  The new state, already validated.
     * @return Result<null> Void on success; a 500 failure on a write error.
     *                      Matching no row is *not* a failure — callers that
     *                      care load the product first.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function update(IProduct $product): Result;

    /**
     * Removes a product.
     *
     * Cargo lines referencing it go with it — `container_items` cascades on
     * delete — so a product still loaded into a container can be removed, and
     * the containers holding it lose those lines.
     *
     * @param  string  $id  Base62 id.
     * @return Result<null> Void on success; a 500 failure on a write error.
     *                      Matching no row is *not* a failure — the use case
     *                      loads the product first, and that is where the 404
     *                      comes from.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function delete(string $id): Result;
}
