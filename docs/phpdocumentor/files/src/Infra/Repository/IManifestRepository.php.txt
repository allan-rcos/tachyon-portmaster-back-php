namespace Infra\Repository;

use Domain\Models\IManifestCargo;
use Shared\Exceptions\Result;

/**
 * Persistence for what a container carries, and for what happened to it.
 *
 * Two tables behind one contract: the cargo lines that make up a manifest, keyed
 * by container and product together, and the append-only telemetry log beside
 * them. Both cascade from `containers`, so deleting a container through
 * {@see IContainerRepository} takes its manifest and its history with it and
 * nothing here needs calling first.
 *
 * Stores what the table module already validated — capacity, weight and the
 * rest are decided before anything reaches here. Every method enlists in the
 * caller's open transaction.
 *
 * @see IManifestCargo A single cargo line.
 * @see \Domain\TableModules\IManifestTM Validates before anything reaches here.
 * @see \Infra\Repository\Interno\SqlManifestRepository The implementation.
 *
 * @license {@link https://opensource.org/licenses/MIT MIT}
 * @copyright 2026 Tachyon
 * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
 *
 * @since 0.0.1
 *
 * @version 0.0.1
 */
interface IManifestRepository
{
    /**
     * Loads one product's cargo line from a container.
     *
     * @param  string  $containerId  Base62 id of the container.
     * @param  string  $productId  Base62 id of the product.
     * @return Result<IManifestCargo|null> The line, or null when the container
     *                                     does not carry that product. An absent
     *                                     line is a success, not a 404 — callers
     *                                     loading a line to add to it need to
     *                                     tell "not carried yet" from "the read
     *                                     broke", which is the 500.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function findCargo(string $containerId, string $productId): Result;

    /**
     * Writes the product's cargo line, replacing the previous one if there is
     * any.
     *
     * Deletes then inserts rather than issuing an upsert, so the stored line is
     * exactly what was passed in and no previous column survives. Both
     * statements are in the caller's transaction, so the line is never
     * observably missing.
     *
     * @param  IManifestCargo  $cargo  Carries both ids and the resulting
     *                                 quantity and weight, already validated.
     * @return Result<null> Void on success; a 500 failure on a write error,
     *                      which includes a container or product that no longer
     *                      exists — the foreign keys reject it.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function upsertCargo(IManifestCargo $cargo): Result;

    /**
     * Removes one product's cargo line from a container.
     *
     * @param  string  $containerId  Base62 id of the container.
     * @param  string  $productId  Base62 id of the product.
     * @return Result<null> Void on success; a 500 failure on a write error.
     *                      Matching no line is *not* a failure.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function deleteCargo(string $containerId, string $productId): Result;

    /**
     * Empties a container's manifest, every line at once.
     *
     * The telemetry log is left alone — what the container carried is gone, the
     * record that it carried it is not.
     *
     * @param  string  $containerId  Base62 id of the container.
     * @return Result<null> Void on success; a 500 failure on a write error. An
     *                      already-empty manifest is *not* a failure.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function clearManifest(string $containerId): Result;

    /**
     * Appends an entry to the container's telemetry log.
     *
     * Append-only: the id is the table's auto-increment and the timestamp is the
     * database's `NOW()`, so neither is the caller's to supply and entries are
     * stamped by the server clock rather than by whichever worker handled the
     * request.
     *
     * @param  string  $containerId  Base62 id of the container.
     * @param  string  $event  Slug of a registered telemetry event.
     * @param  string|null  $description  Free text, or null when the slug says
     *                                    enough on its own.
     * @return Result<null> Void on success; a 500 failure on a write error.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function insertTelemetry(string $containerId, string $event, ?string $description): Result;
}
