declare(strict_types=1);

namespace Infra\Repository;

use Domain\Models\ITelemetryEvent;
use Ds\Seq;
use Shared\Exceptions\Result;

/**
 * The telemetry-event registry: the runtime catalogue of every event type the
 * application layer declared.
 *
 * Same metadata contract as {@see IPermissionRepository}, over the
 * `telemetry_events` table — see there for why a catalogue derivable from the
 * code is a registry rather than a persistence repository, and for why the reads
 * return bare values instead of {@see Result}.
 *
 * One difference worth knowing: unlike a marker, which stores its group's index,
 * a telemetry log row stores the event **slug**. The index assigned here is a
 * registry key and nothing else, so a log written before a restart still reads
 * back correctly even if the events happen to register in another order.
 *
 * @see ITelemetryEvent What is registered.
 * @see IManifestRepository::insertTelemetry() Writes the log rows these describe.
 * @see \Infra\Repository\Interno\TelemetryEventRegistry The implementation.
 * @see docs/adr/0002-metadata-registries-in-the-database.md Why this lives in the database.
 *
 * @license {@link https://opensource.org/licenses/MIT MIT}
 * @copyright 2026 Tachyon
 * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
 *
 * @since 0.0.1
 *
 * @version 0.0.1
 */
interface ITelemetryEventRepository
{
    /**
     * Registers an event type and returns it carrying its assigned registry
     * index.
     *
     * Idempotent by slug, which is what lets every worker declare the same
     * events at boot without duplicating them.
     *
     * @param  ITelemetryEvent  $event  Built by the table module, with `id = 0`.
     * @return Result<ITelemetryEvent> The registered event, carrying its index;
     *                                 a 500 failure when the write fails.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function add(ITelemetryEvent $event): Result;

    /**
     * The registered event type for a slug.
     *
     * @param  string  $slug  What the log row stores.
     * @return ?ITelemetryEvent Null when the slug was never registered.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function getBySlug(string $slug): ?ITelemetryEvent;

    /**
     * The registered event type for a registry index.
     *
     * @param  int  $id  The index assigned at registration.
     * @return ?ITelemetryEvent Null when the index is unknown.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function getById(int $id): ?ITelemetryEvent;

    /**
     * Every registered event type, in registration order.
     *
     * @return Seq<ITelemetryEvent> The whole catalogue; empty before any worker
     *                              has registered anything.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function all(): Seq;

    /**
     * Whether a slug has been registered.
     *
     * @param  string  $slug  What the log row stores.
     * @return bool True when it is in the catalogue.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function has(string $slug): bool;
}
