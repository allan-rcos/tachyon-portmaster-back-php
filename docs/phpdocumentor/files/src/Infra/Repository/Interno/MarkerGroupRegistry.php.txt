declare(strict_types=1);

namespace Infra\Repository\Interno;

use Domain\Models\IMarkerGroup;
use Domain\Models\Internal\MarkerGroup;
use Ds\Seq;
use Infra\Repository\IMarkerGroupRepository;
use Shared\Exceptions\Result;

/**
 * {@see IMarkerGroupRepository} over the `marker_groups` table, filled at
 * WorkerStart.
 *
 * Everything but the three hooks at the bottom is inherited: this class only
 * names its table, its family and its concrete type, and re-types the base's
 * reads to {@see IMarkerGroup} so callers are not handed a bare `object`.
 *
 * {@see \Infra\Repository\Interno\SqlMarkerRepository} calls {@see getBySlug()}
 * on both of its operations, which makes this the hottest of the three
 * registries.
 *
 * @see IMarkerGroupRepository The contract this implements.
 * @see SqlMetadataRegistry Where the behaviour actually lives.
 *
 * @license {@link https://opensource.org/licenses/MIT MIT}
 * @copyright 2026 Tachyon
 * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
 *
 * @since 0.0.1
 *
 * @version 0.0.1
 *
 * @extends SqlMetadataRegistry<IMarkerGroup>
 *
 * @internal
 */
final class MarkerGroupRegistry extends SqlMetadataRegistry implements IMarkerGroupRepository
{
    /**
     * Registers the group's slug, which is the whole of it.
     *
     * @param  IMarkerGroup  $group  Built by the table module, with `id = 0`;
     *                               only its slug is read.
     * @return Result<IMarkerGroup> The registered group carrying its index; a
     *                              500 failure when the write fails.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     */
    public function add(IMarkerGroup $group): Result
    {
        return $this->register($group->slug);
    }

    /**
     * The registered group for a slug.
     *
     * @param  string  $slug  Name the feature declared the group under.
     * @return ?IMarkerGroup Null when unknown, or the read failed.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     */
    public function getBySlug(string $slug): ?IMarkerGroup
    {
        return $this->find($slug);
    }

    /**
     * The registered group for a registry index.
     *
     * @param  int  $id  The index assigned at registration — what the marker row
     *                   stores.
     * @return ?IMarkerGroup Null when unknown, or the read failed.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     */
    public function getById(int $id): ?IMarkerGroup
    {
        return $this->findById($id);
    }

    /**
     * Every registered group, in registration order.
     *
     * @return Seq<IMarkerGroup> The whole catalogue; empty before any worker has
     *                           registered anything.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     */
    public function all(): Seq
    {
        return $this->listAll();
    }

    /**
     * Rebuilds a {@see MarkerGroup} from a stored row.
     *
     * @param  string  $slug  The stored slug.
     * @param  int  $id  The stored registry index.
     * @return IMarkerGroup The concrete group.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     */
    protected function hydrate(string $slug, int $id): IMarkerGroup
    {
        return new MarkerGroup($slug, $id);
    }

    /**
     * Names this family in error messages and in the log channel.
     *
     * @return string `marker group`.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     */
    protected function label(): string
    {
        return 'marker group';
    }

    /**
     * The table backing this family.
     *
     * @return string `marker_groups`.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     */
    protected function table(): string
    {
        return 'marker_groups';
    }
}
