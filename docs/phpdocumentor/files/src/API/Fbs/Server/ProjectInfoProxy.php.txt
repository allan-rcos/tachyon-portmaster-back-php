declare(strict_types=1);

namespace API\Fbs\Server;

use API\Fbs\Contracts\CoercesJson;
use API\Fbs\Contracts\IFbsProxy;
use API\Http\ContentKind;
use API\Http\RequestAttributes;
use Google\FlatBuffers\ByteBuffer;
use Google\FlatBuffers\FlatbufferBuilder;
use OpenSwoole\Core\Psr\Stream;
use Psr\Http\Message\StreamInterface;

/**
 * JSON/binary-aware proxy around the generated {@see ProjectInfo} table.
 *
 * Replaces the old `ProjectInfoDTO` + hand-rolled `json_encode` in the
 * controller: `GET /info` is now a declared, published contract like every other
 * endpoint, and it honours content negotiation for free.
 *
 * **The proxy shape**, written out once here because this is the simplest one —
 * flat, all scalars, no child tables or vectors — and referenced by the other
 * thirty-five rather than restated. Every proxy is the same eight methods over
 * a different field list: a constructor of promoted, defaulted properties;
 * `buildInto()`/`toBinary()` out to binary; `fromTable()`/`fromBinary()` back
 * in; `jsonUnserialize()`/`jsonSerialize()` the JSON pair; and
 * `fromStream()`/`toStream()`, which pick between the two by reading the
 * negotiated {@see ContentKind}. Only the field list differs, so the other
 * proxies document what is particular to them and point here for the rest.
 *
 * Two things the shape settles, and which no proxy repeats: strings are
 * nullable and numbers default to zero, so an absent field is representable
 * rather than an error; and the JSON keys are the **schema's** snake_case names,
 * not the PHP property names, so `memoryUsageMb` is `memory_usage_mb` on the
 * wire.
 *
 * @see IFbsProxy The contract, and why none of this validates.
 * @see CoercesJson Supplies the JSON narrowing helpers.
 *
 * @license {@link https://opensource.org/licenses/MIT MIT}
 * @copyright 2026 Tachyon
 * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
 *
 * @since 0.0.1
 *
 * @version 0.0.1
 */
final class ProjectInfoProxy extends ProjectInfo implements IFbsProxy
{
    use CoercesJson;

    /**
     * Every field defaults, so `new static()` is a valid empty message — which
     * is what an unparseable body hydrates to.
     *
     * @param  string|null  $name  Project name.
     * @param  string|null  $version  Published version.
     * @param  string|null  $environment  Which environment the server booted as.
     * @param  string|null  $runtime  PHP version and server runtime.
     * @param  float  $memoryUsageMb  This worker's current memory use.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     */
    public function __construct(
        public ?string $name = null,
        public ?string $version = null,
        public ?string $environment = null,
        public ?string $runtime = null,
        public float $memoryUsageMb = 0.0,
    ) {
    }

    /**
     * {@inheritDoc}
     *
     * Every string is created on the builder first and a null field becomes
     * offset `0`, which FlatBuffers reads back as absent.
     *
     * @param  FlatbufferBuilder  $builder  The builder to append to.
     * @return int This table's offset within it.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function buildInto(FlatbufferBuilder $builder): int
    {
        $name = $this->name !== null ? $builder->createString($this->name) : 0;
        $version = $this->version !== null ? $builder->createString($this->version) : 0;
        $environment = $this->environment !== null ? $builder->createString($this->environment) : 0;
        $runtime = $this->runtime !== null ? $builder->createString($this->runtime) : 0;

        return ProjectInfo::createProjectInfo(
            $builder,
            $name,
            $version,
            $environment,
            $runtime,
            $this->memoryUsageMb,
        );
    }

    /**
     * {@inheritDoc}
     *
     * @return string A finished, size-prefixed buffer.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function toBinary(): string
    {
        $builder = new FlatbufferBuilder(0);
        $builder->finish($this->buildInto($builder));

        return $builder->sizedByteArray();
    }

    /**
     * Copies a generated table's fields into a proxy.
     *
     * Not part of {@see IFbsProxy}: it takes the generated type, which the
     * contract cannot name. It exists so a parent proxy can rebuild this one
     * out of a decoded buffer without going through binary a second time.
     *
     * @param  ProjectInfo  $table  The generated table to read.
     * @return static The hydrated proxy.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public static function fromTable(ProjectInfo $table): static
    {
        return new static(
            name: $table->getName(),
            version: $table->getVersion(),
            environment: $table->getEnvironment(),
            runtime: $table->getRuntime(),
            memoryUsageMb: (float) $table->getMemoryUsageMb(),
        );
    }

    /**
     * {@inheritDoc}
     *
     * @param  string  $binary  A buffer produced against the same schema.
     * @return static The hydrated proxy.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public static function fromBinary(string $binary): static
    {
        return static::fromTable(ProjectInfo::getRootAsProjectInfo(ByteBuffer::wrap($binary)));
    }

    /**
     * {@inheritDoc}
     *
     * @param  array<string, mixed>  $data  A `json_decode(..., true)` result.
     * @return static The hydrated proxy.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public static function jsonUnserialize(array $data): static
    {
        return new static(
            name: self::jsonNullableString($data, 'name'),
            version: self::jsonNullableString($data, 'version'),
            environment: self::jsonNullableString($data, 'environment'),
            runtime: self::jsonNullableString($data, 'runtime'),
            memoryUsageMb: self::jsonFloat($data, 'memory_usage_mb'),
        );
    }

    /**
     * {@inheritDoc}
     *
     * @return array<string, mixed> Ready for `json_encode()`.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function jsonSerialize(): array
    {
        return [
            'name' => $this->name,
            'version' => $this->version,
            'environment' => $this->environment,
            'runtime' => $this->runtime,
            'memory_usage_mb' => $this->memoryUsageMb,
        ];
    }

    /**
     * {@inheritDoc}
     *
     * Binary is the fallback: only an explicitly JSON request body takes the
     * JSON path, matching {@see ContentKind::fromContentType()}.
     *
     * @param  StreamInterface  $body  The request body.
     * @return static The hydrated proxy; an empty one when the body does not
     *                parse.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public static function fromStream(StreamInterface $body): static
    {
        $raw = (string) $body;

        if (RequestAttributes::RequestContentKind->read() === ContentKind::Json) {
            $decoded = json_decode($raw, true);

            return static::jsonUnserialize(is_array($decoded) ? $decoded : []);
        }

        return static::fromBinary($raw);
    }

    /**
     * {@inheritDoc}
     *
     * @return StreamInterface The response body.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function toStream(): StreamInterface
    {
        $payload = RequestAttributes::ResponseContentKind->read() === ContentKind::Json
            ? (string) json_encode($this)
            : $this->toBinary();

        return Stream::streamFor($payload);
    }
}
