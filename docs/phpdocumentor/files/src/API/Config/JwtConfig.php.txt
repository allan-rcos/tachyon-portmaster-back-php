declare(strict_types=1);

namespace API\Config;

/**
 * Session token settings.
 *
 * Lives in the **API layer** because JSON Web Tokens are a presentation-level
 * transport decision: how this application happens to carry a session over
 * HTTP. Nothing in `App`, `Domain` or `Infra` knows the session is a JWT, or a
 * cookie, or a token at all — they only ever see a
 * {@see \App\Context\UserContext}.
 *
 * The access token travels in an HTTP-only cookie (never the `Authorization`
 * header), so it is unreachable from JavaScript; the protection comes from the
 * cookie flags: `HttpOnly` (XSS), `Secure` (HTTPS only) and `SameSite` (CSRF).
 *
 * Resolved from the environment at bootstrap (see
 * {@see \API\Bootstrap\DotEnvStarter}) and injected, so no service reads env.
 *
 * @see \API\Auth\Interno\FirebaseJwtTokenService What signs and verifies with these.
 * @see \API\Http\AuthCookie What applies the cookie flags.
 *
 * @license {@link https://opensource.org/licenses/MIT MIT}
 * @copyright 2026 Tachyon
 * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
 *
 * @since 0.0.1
 *
 * @version 0.0.1
 */
final readonly class JwtConfig
{
    /**
     * @param  string  $secret  Signing key for the HMAC algorithm. The empty
     *                          default is a working local configuration and an
     *                          unusable production one.
     * @param  int  $ttlSeconds  How long an access token stays valid. Short by
     *                           design; the refresh token below is what avoids
     *                           the logout.
     * @param  string  $issuer  Value of the `iss` claim, checked on verify.
     * @param  string  $algorithm  JWS algorithm name.
     * @param  string  $cookieName  Cookie carrying the access token. Must match
     *                              the `CookieAuth` scheme documented in
     *                              swagger.json.
     * @param  bool  $cookieSecure  Whether both cookies carry `Secure`, which
     *                              confines them to HTTPS. Off by default so
     *                              local HTTP works.
     * @param  string  $cookieSameSite  `SameSite` attribute for both cookies.
     * @param  string  $refreshCookieName  Cookie carrying the refresh token.
     *                                     Long-lived by design: it is what lets
     *                                     the short-lived access token expire
     *                                     quickly without logging the user out.
     *                                     The token itself is signed and carries
     *                                     the same principal — the `HttpOnly`
     *                                     flag, not opacity, is what keeps it
     *                                     away from scripts.
     * @param  int  $refreshTtlSeconds  How long a refresh token stays valid.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     */
    public function __construct(
        public string $secret = '',
        public int $ttlSeconds = 3600,
        public string $issuer = 'tachyon/portmaster',
        public string $algorithm = 'HS256',
        public string $cookieName = 'auth_token',
        public bool $cookieSecure = false,
        public string $cookieSameSite = 'Strict',
        public string $refreshCookieName = 'refresh_token',
        public int $refreshTtlSeconds = 1209600, // 14 days
    ) {
    }
}
