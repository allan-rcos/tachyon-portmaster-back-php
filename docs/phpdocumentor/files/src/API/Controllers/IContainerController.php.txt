declare(strict_types=1);

namespace API\Controllers;

use Psr\Http\Message\ResponseInterface;
use Psr\Http\Message\ServerRequestInterface;

/**
 * Container endpoints (`/containers`).
 *
 * CRUD plus the two lifecycle transitions. `seal` and `dispatch` are POSTs to
 * their own paths rather than a status field on {@see update()}: the moves are
 * one-way and refusable, and a 409 for "this container cannot be sealed yet"
 * belongs to an action, not to a field.
 *
 * @see IProductController The contract shape these follow.
 * @see \Domain\Enums\ContainerStatus The transitions the last two perform.
 *
 * @license {@link https://opensource.org/licenses/MIT MIT}
 * @copyright 2026 Tachyon
 * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
 *
 * @since 0.0.1
 *
 * @version 0.0.1
 */
interface IContainerController
{
    /**
     * `GET /containers` — a keyset page of containers.
     *
     * @param  ServerRequestInterface  $request  The incoming HTTP request.
     * @return ResponseInterface The outgoing HTTP response.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function list(ServerRequestInterface $request): ResponseInterface;

    /**
     * `POST /containers` — registers a container. Answers 201.
     *
     * @param  ServerRequestInterface  $request  The incoming HTTP request.
     * @return ResponseInterface The outgoing HTTP response.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function create(ServerRequestInterface $request): ResponseInterface;

    /**
     * `GET /containers/summary` — a keyset page of containers with their fill
     * and cargo rolled up.
     *
     * Its route **must** be registered before `/containers/{id}`, or the id
     * pattern swallows the word `summary`.
     *
     * @param  ServerRequestInterface  $request  The incoming HTTP request.
     * @return ResponseInterface The outgoing HTTP response.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function summary(ServerRequestInterface $request): ResponseInterface;

    /**
     * `GET /containers/{id}` — one container, or 404.
     *
     * @param  ServerRequestInterface  $request  The incoming HTTP request.
     * @return ResponseInterface The outgoing HTTP response.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function get(ServerRequestInterface $request): ResponseInterface;

    /**
     * `PUT /containers/{id}` — replaces a container's fields. Its status is not
     * among them; see {@see seal()} and {@see dispatch()}.
     *
     * @param  ServerRequestInterface  $request  The incoming HTTP request.
     * @return ResponseInterface The outgoing HTTP response.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function update(ServerRequestInterface $request): ResponseInterface;

    /**
     * `DELETE /containers/{id}` — removes a container. Answers 204.
     *
     * @param  ServerRequestInterface  $request  The incoming HTTP request.
     * @return ResponseInterface The outgoing HTTP response.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function delete(ServerRequestInterface $request): ResponseInterface;

    /**
     * `POST /containers/{id}/seal` — closes a container to further cargo.
     * Answers 409 when the transition is refused.
     *
     * @param  ServerRequestInterface  $request  The incoming HTTP request.
     * @return ResponseInterface The outgoing HTTP response.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function seal(ServerRequestInterface $request): ResponseInterface;

    /**
     * `POST /containers/{id}/dispatch` — sends a sealed container into transit,
     * which is terminal. Answers 409 when the transition is refused.
     *
     * @param  ServerRequestInterface  $request  The incoming HTTP request.
     * @return ResponseInterface The outgoing HTTP response.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function dispatch(ServerRequestInterface $request): ResponseInterface;
}
