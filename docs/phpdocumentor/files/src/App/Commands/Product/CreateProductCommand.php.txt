declare(strict_types=1);

namespace App\Commands\Product;

use App\Context\UserContext;

use Domain\Enums\RiskClass;

/**
 * The input to {@see \App\Services\ICreateProductUseCase}: who is asking, and
 * what they want created.
 *
 * **The shape every command in this layer follows.** A `$context` first, then
 * the fields — a plain immutable carrier with no behaviour and no validation.
 * Deciding whether the fields are acceptable belongs to the table module, and
 * deciding whether the caller may belongs to the use case; a command that
 * checked either would put the rule in the wrong place and let a caller
 * construct their way around it.
 *
 * The fields are already typed — `float`, {@see RiskClass} — so parsing has
 * happened at the edge and the application layer never sees a raw string.
 *
 * @see \App\Services\ICreateProductUseCase What consumes it.
 * @see UserContext Who is asking.
 *
 * @license {@link https://opensource.org/licenses/MIT MIT}
 * @copyright 2026 Tachyon
 * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
 *
 * @since 0.0.1
 *
 * @version 0.0.1
 */
final readonly class CreateProductCommand
{
    /**
     * @param  UserContext  $context  The caller, established at the edge from
     *                                their session.
     * @param  string  $name  Commercial name; the table module decides whether
     *                        it is acceptable.
     * @param  float  $density  Kilograms per litre.
     * @param  RiskClass  $riskClass  Already resolved to the enum.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     */
    public function __construct(
        public UserContext $context,
        public string $name,
        public float $density,
        public RiskClass $riskClass,
    ) {
    }
}
