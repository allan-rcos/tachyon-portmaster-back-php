declare(strict_types=1);

namespace App\Services;

use App\Commands\Manifest\UnloadItemCommand;
use Domain\Models\IContainer;
use Shared\Exceptions\Result;

/**
 * Removes a quantity of a product from a container.
 *
 * The inverse of {@see ILoadItemUseCase}, and identically shaped: load the
 * container, the product and the existing cargo line, let the table module work
 * out the new state, then persist the line, the container's weight and a
 * telemetry entry inside one boundary.
 *
 * Guarded by `manifest:unload` — a separate privilege from loading, so a role
 * can be allowed to fill a container without being allowed to empty it.
 *
 * @see UnloadItemCommand What it takes.
 * @see \App\Services\Interno\UnloadItemUseCase The implementation.
 * @see ILoadItemUseCase The inverse.
 *
 * @license {@link https://opensource.org/licenses/MIT MIT}
 * @copyright 2026 Tachyon
 * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
 *
 * @since 0.0.1
 *
 * @version 0.0.1
 */
interface IUnloadItemUseCase
{
    /**
     * Unloads the quantity the command names.
     *
     * @param  UnloadItemCommand  $command  Carries the caller, both ids and the
     *                                      quantity.
     * @return Result<IContainer> The container's new state after the unload; 404
     *                            when the container, product or cargo line is
     *                            missing, 409 or 422 when the domain refuses the
     *                            amount.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function execute(UnloadItemCommand $command): Result;
}
