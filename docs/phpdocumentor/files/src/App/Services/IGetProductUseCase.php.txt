declare(strict_types=1);

namespace App\Services;

use App\Queries\Product\GetProductQuery;
use Infra\Query\Product\ProductViewItem;
use Shared\Exceptions\Result;

/**
 * Reads one product by id.
 *
 * One of the single-read use cases, all of which share the shape documented on
 * {@see \App\Services\Interno\GetProductUseCase}: authorize, run the DQL, and
 * turn an absent row into a 404.
 *
 * Guarded by `product:read`, shared with {@see IListProductsUseCase}.
 *
 * @see GetProductQuery What it takes.
 * @see \App\Services\Interno\GetProductUseCase The implementation, and the shape.
 *
 * @license {@link https://opensource.org/licenses/MIT MIT}
 * @copyright 2026 Tachyon
 * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
 *
 * @since 0.0.1
 *
 * @version 0.0.1
 */
interface IGetProductUseCase
{
    /**
     * Reads the product the query names.
     *
     * @param  GetProductQuery  $query  Carries the caller and the id.
     * @return Result<ProductViewItem> The product, or 404 when not found; a 403
     *                                 or 500 otherwise.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function execute(GetProductQuery $query): Result;
}
