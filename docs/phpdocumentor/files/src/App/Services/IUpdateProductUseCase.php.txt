declare(strict_types=1);

namespace App\Services;

use App\Commands\Product\UpdateProductCommand;
use Domain\Models\IProduct;
use Shared\Exceptions\Result;

/**
 * Replaces a product's fields.
 *
 * **The shape every update use case in this layer follows**: authorize, begin,
 * *load*, rebuild through the table module, persist, commit. The load is what
 * distinguishes it from a create — the existing record has to be fetched so an
 * absent one is a 404, and so the table module rebuilds from real state rather
 * than from whatever the caller sent.
 *
 * Guarded by `product:update`.
 *
 * @see UpdateProductCommand What it takes.
 * @see \App\Services\Interno\UpdateProductUseCase The implementation.
 * @see ICreateProductUseCase The create counterpart.
 *
 * @license {@link https://opensource.org/licenses/MIT MIT}
 * @copyright 2026 Tachyon
 * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
 *
 * @since 0.0.1
 *
 * @version 0.0.1
 */
interface IUpdateProductUseCase
{
    /**
     * Replaces the product the command names.
     *
     * @param  UpdateProductCommand  $command  Carries the caller, the id and
     *                                         every field.
     * @return Result<IProduct> The updated product; 404 when it does not exist,
     *                          422 when the table module refused the fields, a
     *                          403 or 500 otherwise.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function execute(UpdateProductCommand $command): Result;
}
