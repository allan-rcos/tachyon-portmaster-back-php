declare(strict_types=1);

namespace App\Services;

use App\Commands\Product\DeleteProductCommand;
use Shared\Exceptions\Result;

/**
 * Removes a product.
 *
 * Follows the write shape documented on
 * {@see \App\Services\Interno\CreateProductUseCase}, with a load in place of the
 * build: the product is fetched first so that a missing one is a 404 rather than
 * a silent no-op, since the repository does not treat "matched no row" as a
 * failure.
 *
 * Guarded by `product:delete`.
 *
 * @see DeleteProductCommand What it takes.
 * @see \App\Services\Interno\DeleteProductUseCase The implementation.
 * @see \App\Services\Interno\CreateProductUseCase The shape.
 *
 * @license {@link https://opensource.org/licenses/MIT MIT}
 * @copyright 2026 Tachyon
 * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
 *
 * @since 0.0.1
 *
 * @version 0.0.1
 */
interface IDeleteProductUseCase
{
    /**
     * Removes the product the command names.
     *
     * @param  DeleteProductCommand  $command  Carries the caller and the id.
     * @return Result<null> Void on success, 404 when the product does not exist;
     *                      a 403 or 500 otherwise.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function execute(DeleteProductCommand $command): Result;
}
