declare(strict_types=1);

namespace App\Services;

use App\Queries\Metrics\GetMetricsQuery;
use Infra\Query\Metrics\MetricsView;
use Shared\Exceptions\Result;

/**
 * Reads the yard-wide metrics snapshot.
 *
 * Follows the list-read shape documented on
 * {@see \App\Services\Interno\ListProductsUseCase}: authorize, run the DQL,
 * return what comes back. No 404 — the snapshot always exists, even of an empty
 * yard.
 *
 * Guarded by `metrics:read`.
 *
 * @see GetMetricsQuery What it takes.
 * @see \App\Services\Interno\GetMetricsUseCase The implementation.
 * @see \App\Services\Interno\ListProductsUseCase The shape.
 *
 * @license {@link https://opensource.org/licenses/MIT MIT}
 * @copyright 2026 Tachyon
 * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
 *
 * @since 0.0.1
 *
 * @version 0.0.1
 */
interface IGetMetricsUseCase
{
    /**
     * Reads the snapshot.
     *
     * @param  GetMetricsQuery  $query  Carries the caller, and nothing else.
     * @return Result<MetricsView> The snapshot as of the moment the query ran; a
     *                             403 or 500 failure.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function execute(GetMetricsQuery $query): Result;
}
