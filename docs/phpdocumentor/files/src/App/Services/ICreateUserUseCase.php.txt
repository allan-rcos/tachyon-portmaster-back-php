declare(strict_types=1);

namespace App\Services;

use App\Commands\User\CreateUserCommand;
use Domain\Models\IUser;
use Shared\Exceptions\Result;

/**
 * Creates a user and assigns their roles in one boundary.
 *
 * Follows the write shape documented on
 * {@see \App\Services\Interno\CreateProductUseCase}, with an extra persist step:
 * the user row and the role assignments are two writes, and the transaction is
 * what makes them one operation.
 *
 * Guarded by `user:create`.
 *
 * @see CreateUserCommand What it takes.
 * @see \App\Services\Interno\CreateUserUseCase The implementation.
 * @see \App\Services\Interno\CreateProductUseCase The shape.
 *
 * @license {@link https://opensource.org/licenses/MIT MIT}
 * @copyright 2026 Tachyon
 * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
 *
 * @since 0.0.1
 *
 * @version 0.0.1
 */
interface ICreateUserUseCase
{
    /**
     * Creates the user described by the command.
     *
     * @param  CreateUserCommand  $command  Carries the caller, the profile, the
     *                                      initial password and the role ids.
     * @return Result<IUser> The created user, or 409 when the e-mail is taken; a
     *                       403, 422 or 500 otherwise.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function execute(CreateUserCommand $command): Result;
}
