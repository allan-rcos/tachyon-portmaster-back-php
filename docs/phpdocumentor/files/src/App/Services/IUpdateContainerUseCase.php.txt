declare(strict_types=1);

namespace App\Services;

use App\Commands\Container\UpdateContainerCommand;
use Domain\Models\IContainer;
use Shared\Exceptions\Result;

/**
 * Changes a container's capacity.
 *
 * Follows the update shape documented on {@see IUpdateProductUseCase}. Capacity
 * is the only editable field; status moves through the transition use cases.
 *
 * Guarded by `container:update`.
 *
 * @see UpdateContainerCommand What it takes.
 * @see \App\Services\Interno\UpdateContainerUseCase The implementation.
 * @see IUpdateProductUseCase The shape.
 *
 * @license {@link https://opensource.org/licenses/MIT MIT}
 * @copyright 2026 Tachyon
 * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
 *
 * @since 0.0.1
 *
 * @version 0.0.1
 */
interface IUpdateContainerUseCase
{
    /**
     * Changes the capacity of the container the command names.
     *
     * @param  UpdateContainerCommand  $command  Carries the caller, the id and
     *                                           the new capacity.
     * @return Result<IContainer> The updated container, or 404 when not found;
     *                            422 when the domain refuses the figure, a 403
     *                            or 500 otherwise.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     *
     * @api
     */
    public function execute(UpdateContainerCommand $command): Result;
}
