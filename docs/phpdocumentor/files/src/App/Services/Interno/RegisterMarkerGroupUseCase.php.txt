declare(strict_types=1);

namespace App\Services\Interno;

use App\Commands\Marker\RegisterMarkerGroupCommand;
use App\Services\IRegisterMarkerGroupUseCase;
use Domain\Models\IMarkerGroup;
use Domain\TableModules\IMarkerGroupTM;
use Infra\Repository\IMarkerGroupRepository;
use RuntimeException;
use Shared\Exceptions\Leaf;

/**
 * Encapsulates the marker-group declaration handshake, mirroring
 * {@see RegisterPermissionUseCase}: the domain table module validates and builds
 * the group, then the infra registry stores it and hands back its index.
 *
 * Throws rather than returning a failed {@see \Shared\Exceptions\Result}, for
 * the reason its twin does: a server that booted without a group it needs would
 * answer every request depending on it with a 404.
 *
 * @see IRegisterMarkerGroupUseCase The contract this implements, and the reasoning.
 * @see RegisterPermissionUseCase The permission twin.
 * @uses IMarkerGroupTM Validates the slug and builds the group.
 * @uses IMarkerGroupRepository Stores it and hands back its index.
 *
 * @license {@link https://opensource.org/licenses/MIT MIT}
 * @copyright 2026 Tachyon
 * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
 *
 * @since 0.0.1
 *
 * @version 0.0.1
 *
 * @internal
 */
final readonly class RegisterMarkerGroupUseCase implements IRegisterMarkerGroupUseCase
{
    /**
     * Takes no unit of work: the registry leases its own connection, having no
     * request boundary to enlist in.
     *
     * @param  IMarkerGroupTM  $markerGroupTM  Validates the slug's shape.
     * @param  IMarkerGroupRepository  $groups  The runtime registry.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     */
    public function __construct(
        private IMarkerGroupTM $markerGroupTM,
        private IMarkerGroupRepository $groups,
    ) {
    }

    /**
     * @inheritDoc
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     */
    public function execute(RegisterMarkerGroupCommand $command): string
    {
        $built = $this->markerGroupTM->create($command->slug);

        if (!$built->isSuccess()) {
            throw new RuntimeException($this->reason($built->getErrorId(), $command->slug));
        }

        /** @var IMarkerGroup $group */
        $group = $built->getValue();

        $added = $this->groups->add($group);

        if (!$added->isSuccess()) {
            throw new RuntimeException($this->reason($added->getErrorId(), $command->slug));
        }

        /** @var IMarkerGroup $registered */
        $registered = $added->getValue();

        return $registered->slug;
    }

    /**
     * Turns a failed {@see \Shared\Exceptions\Result} into the exception message
     * to boot-fail with.
     *
     * The `Leaf` carries the failure's detail, and this is the last place it can
     * be read: once the exception is thrown, the id is gone.
     *
     * @param  int  $errorId  The failed result's `Leaf` id.
     * @param  string  $slug  What was being registered, for the message.
     * @return string A message naming the slug and whatever the failure said,
     *                degrading to "unknown error" when the id resolves to
     *                nothing.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     */
    private function reason(int $errorId, string $slug): string
    {
        $context = Leaf::getError($errorId);
        $message = $context !== null ? $context->message : 'unknown error';
        $details = $context?->details?->toArray() ?? [];

        return sprintf(
            'Cannot register marker group "%s": %s%s',
            $slug,
            $message,
            $details === [] ? '' : ' ('.json_encode($details).')',
        );
    }
}
