declare(strict_types=1);

namespace App\Services\Interno;

use App\Context\UserContext;
use App\Services\IValidateSessionUseCase;
use Domain\Models\IRole;
use Domain\Models\Internal\User;
use Domain\Models\IUser;
use Infra\Database\IUnitOfWork;
use Infra\Repository\IRoleRepository;
use Infra\Repository\IUserRepository;
use Shared\Exceptions\Leaf;
use Shared\Exceptions\LeafContext;
use Shared\Exceptions\Result;

/**
 * Re-reads the user behind a session and confirms it is still valid.
 *
 * The same two loads {@see LoginUseCase} performs after judging the password —
 * the user, then their roles — minus the password itself, because the signed
 * token has already stood in for it.
 *
 * Declares no permission and must not: it runs *during* authentication, so there
 * is no established caller for a permission to be checked against. The claims it
 * receives are the thing under suspicion, not evidence of anything.
 *
 * @see IValidateSessionUseCase The contract this implements, and the reasoning.
 * @see LoginUseCase The same loads, at the start of a session.
 * @uses IUnitOfWork The boundary the reads run in.
 * @uses IUserRepository Re-reads the user named by the claims.
 * @uses IRoleRepository Loads their roles as they are now, not as the token says.
 *
 * @license {@link https://opensource.org/licenses/MIT MIT}
 * @copyright 2026 Tachyon
 * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
 *
 * @since 0.0.1
 *
 * @version 0.0.1
 *
 * @internal
 */
final readonly class ValidateSessionUseCase implements IValidateSessionUseCase
{
    /**
     * Takes no registrar: there is no permission to declare.
     *
     * @param  IUnitOfWork  $unitOfWork  The boundary the reads run in.
     * @param  IUserRepository  $users  Re-reads the user.
     * @param  IRoleRepository  $roles  Loads their current roles.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     */
    public function __construct(
        private IUnitOfWork $unitOfWork,
        private IUserRepository $users,
        private IRoleRepository $roles,
    ) {
    }

    /**
     * @inheritDoc
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     */
    public function execute(UserContext $context): Result
    {
        $begin = $this->unitOfWork->begin();
        if (!$begin->isSuccess()) {
            return Result::failure($begin->getErrorId());
        }

        $found = $this->users->findById($context->id);
        if (!$found->isSuccess()) {
            $this->unitOfWork->rollback();

            return self::invalidSession();
        }

        /** @var IUser $user */
        $user = $found->getValue();

        $rolesResult = $this->roles->findByUserId($user->id);
        if (!$rolesResult->isSuccess()) {
            $this->unitOfWork->rollback();

            return Result::failure($rolesResult->getErrorId());
        }

        /** @var list<IRole> $roles */
        $roles = $rolesResult->getValue();

        $commit = $this->unitOfWork->commit();
        if (!$commit->isSuccess()) {
            return Result::failure($commit->getErrorId());
        }

        return Result::success(new User(
            id: $user->id,
            name: $user->name,
            email: $user->email,
            passwordHash: $user->passwordHash,
            roles: $roles,
        ));
    }

    /**
     * The single 401 every failed validation produces.
     *
     * A deleted account and a session that was never valid come back
     * identically, so a caller cannot use the refresh endpoint to learn whether
     * an account still exists.
     *
     * @return Result<never> Always a 401 failure.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     */
    private static function invalidSession(): Result
    {
        return Result::failure(Leaf::newError(new LeafContext(
            message: 'Invalid or expired session.',
            code: 401,
        )));
    }
}
