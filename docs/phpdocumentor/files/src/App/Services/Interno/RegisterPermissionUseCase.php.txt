declare(strict_types=1);

namespace App\Services\Interno;

use App\Commands\Permission\RegisterPermissionCommand;
use App\Services\IRegisterPermissionUseCase;
use Domain\Models\IPermission;
use Domain\TableModules\IPermissionTM;
use Infra\Repository\IPermissionRepository;
use RuntimeException;
use Shared\Exceptions\Leaf;

/**
 * Encapsulates the permission-declaration handshake so no other use case has to
 * know it: the domain table module validates and builds the permission, then the
 * infra registry stores it and hands back its index.
 *
 * The only use case that throws rather than returning a failed {@see Result} —
 * see {@see IRegisterPermissionUseCase} for why a boot-time problem should stop
 * the server instead of becoming a value to branch on.
 *
 * @see IRegisterPermissionUseCase The contract this implements, and the reasoning.
 * @see RegisterMarkerGroupUseCase The marker-group twin.
 * @uses IPermissionTM Validates the slug and builds the permission.
 * @uses IPermissionRepository Stores it and hands back its index.
 *
 * @license {@link https://opensource.org/licenses/MIT MIT}
 * @copyright 2026 Tachyon
 * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
 *
 * @since 0.0.1
 *
 * @version 0.0.1
 *
 * @internal
 */
final readonly class RegisterPermissionUseCase implements IRegisterPermissionUseCase
{
    /**
     * Takes no unit of work: the registry leases its own connection, having no
     * request boundary to enlist in.
     *
     * @param  IPermissionTM  $permissionTM  Validates the slug's shape.
     * @param  IPermissionRepository  $permissions  The runtime registry.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     */
    public function __construct(
        private IPermissionTM $permissionTM,
        private IPermissionRepository $permissions,
    ) {
    }

    /**
     * @inheritDoc
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     */
    public function execute(RegisterPermissionCommand $command): string
    {
        $built = $this->permissionTM->create($command->slug);

        if (!$built->isSuccess()) {
            throw new RuntimeException($this->reason($built->getErrorId(), $command->slug));
        }

        /** @var IPermission $permission */
        $permission = $built->getValue();

        $added = $this->permissions->add($permission);

        if (!$added->isSuccess()) {
            throw new RuntimeException($this->reason($added->getErrorId(), $command->slug));
        }

        /** @var IPermission $registered */
        $registered = $added->getValue();

        return $registered->slug;
    }

    /**
     * Turns a failed {@see \Shared\Exceptions\Result} into the exception message
     * to boot-fail with.
     *
     * The `Leaf` carries the failure's detail, and this is the last place it can
     * be read: once the exception is thrown, the id is gone. That is why the
     * details are flattened into the string rather than referenced.
     *
     * @param  int  $errorId  The failed result's `Leaf` id.
     * @param  string  $slug  What was being registered, for the message.
     * @return string A message naming the slug and whatever the failure said,
     *                degrading to "unknown error" when the id resolves to
     *                nothing.
     *
     * @copyright 2026 Tachyon
     * @author Ricardo Állan Costa <ricardoallancosta@hotmail.com>
     *
     * @since 0.0.1
     *
     * @version 0.0.1
     */
    private function reason(int $errorId, string $slug): string
    {
        $context = Leaf::getError($errorId);
        $message = $context !== null ? $context->message : 'unknown error';
        $details = $context?->details?->toArray() ?? [];

        return sprintf(
            'Cannot register permission "%s": %s%s',
            $slug,
            $message,
            $details === [] ? '' : ' ('.json_encode($details).')',
        );
    }
}
